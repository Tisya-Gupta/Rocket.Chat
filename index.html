// 1. The Store: packages/ui-client/src/hooks/useThemePreview.ts

import type { ThemePreference as ThemeMode } from '@rocket.chat/core-typings';
import { useSyncExternalStore } from 'react';

const listeners = new Set<() => void>();
let preview: ThemeMode | undefined;

export const setThemePreview = (mode: ThemeMode | undefined): void => {
    if (preview === mode) return;
    preview = mode;
    listeners.forEach((listener) => listener());
};

const subscribe = (listener: () => void): (() => void) => {
    listeners.add(listener);
    return () => { listeners.delete(listener); };
};

const getSnapshot = (): ThemeMode | undefined => preview;

export const useThemePreview = (): ThemeMode | undefined =>
    useSyncExternalStore(subscribe, getSnapshot);

// 2. The Resolver: packages/ui-client/src/hooks/useThemeMode.ts

export const useThemeMode = () => {
    const previewMode = useThemePreview();
    const savedMode = useUserPreference<ThemeMode>('themeAppearence') || 'auto';
    const themeMode = previewMode ?? savedMode; // ⭐ Preview takes priority!
    const isDarkMode = useDarkMode(themeMode === 'auto' ? undefined : themeMode === 'dark');

    if (isDarkMode) return 'dark';
    if (themeMode === 'high-contrast') return 'high-contrast';
    return 'light';
};

// 3. Settings View: packages/ui-account/src/accessibility/views/AccessibilityView.tsx

const currentData = watch();
const isThemeDirty = !!dirtyFields.themeAppearence;

// Live preview effect with dirty gate
useEffect(() => {
    setThemePreview(isThemeDirty ? currentData.themeAppearence : undefined);
}, [currentData.themeAppearence, isThemeDirty]);

// Clean up preview if user unmounts/navigates away
useEffect(() => () => setThemePreview(undefined), []);
